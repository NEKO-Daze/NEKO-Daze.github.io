### 1. 什么是线段树

线段树是一种 $OI$ 中常用的数据结构，常用来维护**区间信息**。

线段树可以在 $\mathcal{O(log~n)}$ 的时间复杂度内实现**单点修改，区间修改，区间查询（如区间和、区间最小值、区间最大值等）**。

如图是 $n=10$ 时所构建的线段树。

![](https://cdn.luogu.com.cn/upload/image_hosting/ei2tuzar.png)

线段树的每个结点都会管辖一段区间。若结点 $p$ 所管辖的区间为 $[l,r]$，那么其左右儿子 $p\times2,p\times2+1$ 就会管辖区间 $[l,mid],[mid+1,r]$，并不断往下迭代建树。当 $l=r$ 时，区间缩为一个点，建树停止。

容易发现线段树中最后一层的一些位置是没有被利用上的，即树的最后一层结点在数组中保存的位置是**不连续的**。我们不需要对其做任何操作。因此在开数组的时候，数组长度要不少于 $4\times N$。

### 2. 代码实现

我们会使用递归的方式构建一个线段树，相当于在对其进行 $DFS$。

假设我们需要构建一个线段树，用于存储区间和。为了方便，我们可以用一个结构体封装线段树。

我们在结构体总定义一个数组 `sum`。对于结点 $p$，它所管辖的区间 $[l,r]$ 的区间和存储在 `sum[p]` 中。

```cpp
struct Tree {//线段树是二叉树，若一个结点为p，那么它的左儿子为p*2，右儿子为p*2+1
    #define mid (l + r >> 1) //区间中值mid，等价于 (l+r)/2
    int sum[N<<2]; //sum为区间和
    //...
```

我们用一个函数 `pushup_sum(int p)` 来在后面更新结点 $p$ 的区间和。易知对于结点 $p$，其区间和等于左儿子与右儿子的区间和之和，即 `sum[p] = sum[p * 2] + sum[p * 2 + 1];`。

在建树函数 `build(int p, int l, int r)` 中，我们不断递归新建左子树与右子树，当发现当前子树所管辖区间 $[l,r]$ 缩为一点时，代表已经无法再分，此时区间和即为点的元素值，即 `sum[p]=a[l]`，并立即 `return;` 结束建树。当左子树与右子树建完后，我们对两个子树进行合并，以得到当前结点的值。即调用 `pushup_sum(int p)` 函数。

因此，我们只需要在主函数中调用 `T.build(1,1,n)` 即可建树。

```cpp
    //...
    //p*2等价于p<<1,p*2+1等价于p<<1|1
    void pushup_sum(int p) //sum值更新，p为当前结点编号
    {
        sum[p] = sum[p << 1] + sum[p << 1 | 1];
    }
    void build(int p, int l, int r)//建树
    {//p为当前结点编号，所表示区间为[l,r]
        if (l == r) {sum[p] = a[l];return ;} //当l=r时，区间缩为一个点，停止向下建子树，并把区间最大值和区间和赋值
        build(p << 1, l, mid);//建左子树
        build(p << 1 | 1, mid + 1, r);//建右子树
        pushup_sum(p);//将新建的两个子树的区间合并，确定当前结点的值
    }
    //...
```

当我们要对一个点 $s$ 继续修改操作，将其的值改为 $t$ 时，我们在单点修改函数 `change(int p, int l, int r, int s, int t)` 中完成。我们使用递归来找到线段树中管辖该点的结点（即结点管辖区间为 $[s,s]$ 的）。

若当前结点为 $p$，所管辖区间为 $[l,r]$，目标是将点 $s$ 的值修改为 $t$。

* 若 $l=r$，代表此时已经找到该点，直接进行修改即可。

* 若 $s\le mid$，说明目标点在左子树。

* 否则，目标点在右子树。

最后要记得更新当前结点的值。因为修改后，区间和也会随之改变。

```cpp
    //...
    void change(int p, int l, int r, int s, int t)//单点修改
    {//p为当前结点编号，所表示区间为[l,r]，需要修改编号为s的结点的值，修改为t
        if (l == r)//当l=r时，区间缩为一个点，代表已经找到了这个点，对其值进行修改
        {
            sum[p] = t;maxn[p]=t;
            return ;
        }
        if (s <= mid) change(p << 1, l, mid, s, t);//结点在左子树
        else change(p << 1 | 1, mid + 1, r, s, t);//结点在右子树
        pushup_sum(p);//将修改后的两个子树的区间合并，更新当前结点的值
    }
    //...
```

当我们要查询区间 $[l,r]$ 的区间和时，我们在查询函数 `query_sum(int p, int l, int r, int s, int t)` 中完成。

若当前结点为 $p$，所管辖区间为 $[l,r]$，目标区间为 $[s,t]$。

* 若 $l\ge s$ 且 $r\le t$，代表当前区间被目标区间 $[s,t]$ 所覆盖，我们直接使用这个区间的区间和对答案作贡献。

* 若 $s\le mid$，说明左子树区间与目标区间有交集。此时我们在左子树中找贡献。

* 若 $t>mid$，说明右子树区间与目标区间有交集。此时我们在右子树中找贡献。

```cpp
    //...
    int query_sum(int p, int l, int r, int s, int t)
    {//当前结点编号为p，当前结点所表示的区间为[l,r]，要查询[s,t]的区间和
        if (l >= s && r <= t) return sum[p]; //当前子树所表示区间被[s,t]完全覆盖，直接使用当前结点的值sum[p]
        int ans = 0;
        if (s <= mid) ans = query_sum(p << 1, l, mid, s, t); // 与左子树有交集但非覆盖，在左子树寻找
        if (t > mid) ans += query_sum(p << 1 | 1, mid + 1, r, s, t); // 与右子树有交集但非覆盖，在右子树寻找
        return ans;
    }
    //...
```

总代码如下（可查询区间和与区间最大值，其他功能同理）：

```cpp
#include <bits/stdc++.h>
using namespace std;
#define N 200005
int a[N], n, m;
struct Tree {//线段树是二叉树，若一个结点为p，那么它的左儿子为p*2，右儿子为p*2+1
    #define mid (l + r >> 1) //区间中值mid，等价于 (l+r)/2
    int sum[N<<2], maxn[N<<2]; //sum为区间和，maxn为区间最大值
    //p*2等价于p<<1,p*2+1等价于p<<1|1
    void pushup_sum(int p) //sum值更新，p为当前结点编号
    {
        sum[p] = sum[p << 1] + sum[p << 1 | 1];
    }
    void pushup_max(int p) //maxn值更新，p为当前结点编号
    {
        maxn[p] = max(maxn[p << 1] , maxn[p << 1 | 1]);
    }
    void build(int p, int l, int r)//建树
    {//p为当前结点编号，所表示区间为[l,r]
        if (l == r) {sum[p] = a[l]; maxn[p] = a[l];
         return ;} //当l=r时，区间缩为一个点，停止向下建子树，并把区间最大值和区间和赋值
        build(p << 1, l, mid);//建左子树
        build(p << 1 | 1, mid + 1, r);//建右子树
        pushup_sum(p);pushup_max(p);//将新建的两个子树的区间合并，确定当前结点的值
    }
    void change(int p, int l, int r, int s, int t)//单点修改
    {//p为当前结点编号，所表示区间为[l,r]，需要修改编号为s的结点的值，修改为t
        if (l == r)//当l=r时，区间缩为一个点，代表已经找到了这个点，对其值进行修改
        {
            sum[p] = t;maxn[p]=t;
            return ;
        }
        if (s <= mid) change(p << 1, l, mid, s, t);//结点在左子树
        else change(p << 1 | 1, mid + 1, r, s, t);//结点在右子树
        pushup_sum(p);pushup_max(p);//将修改后的两个子树的区间合并，更新当前结点的值
    }
    int query_sum(int p, int l, int r, int s, int t)
    {//当前结点编号为p，当前结点所表示的区间为[l,r]，要查询[s,t]的区间和
        if (l >= s && r <= t) return sum[p]; //当前子树所表示区间被[s,t]完全覆盖，直接使用当前结点的值sum[p]
        int ans = 0;
        if (s <= mid) ans = query_sum(p << 1, l, mid, s, t); // 与左子树有交集但非覆盖，在左子树寻找
        if (t > mid) ans += query_sum(p << 1 | 1, mid + 1, r, s, t); // 与右子树有交集但非覆盖，在右子树寻找
        return ans;
    }
    int query_max(int p, int l, int r, int s, int t)
    {//当前结点编号为p，当前结点所表示的区间为[l,r]，要查询[s,t]的区间最大值
        if (l >= s && r <= t) return maxn[p]; //当前子树所表示区间被[s,t]完全覆盖，直接使用当前结点的值maxn[p]

        if (s <= mid) ans = query_max(p << 1, l, mid, s, t); // 与左子树有交集但非覆盖，在左子树寻找
        if (t > mid) ans = max(ans, query_max(p << 1 | 1, mid + 1, r, s, t)); // 与右子树有交集但非覆盖，在右子树寻找
        return ans;
    }
}T;
int main()
{
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    T.build(1, 1, n);
    for (int i = 1; i <= m; i++)
    {
        int op, x, y;
        cin >> op >> x >> y;
        if (op == 1) T.change(1, 1, n, x, y);//单点修改
        else cout << T.query_sum(1, 1, n, x, y) << endl;//查询区间和
    }
}
```

接下来我们实现区间修改的功能。

例如对于长度为 $4$ 的线段树，我们要修改区间 $[2,3]$ 的值，那么就要修改两条路径上的值，因此最坏情况下，一个一个修改的复杂度就会达到 $O(n)$，这是不可接受的。

因此我们可以在修改区间 $s,t$ 时，若遇到包含此区间，即 $l\le s\le t\le r$ 的结点，那么我们直接对其进行修改并停止向下遍历，并做上一个标记，表示：我把这个点的值改了，但它的子结点的值我还没改，等会再改……

之后要在此节点往下遍历的时候，发现这个节点，就将两个子结点的值修改了，同时下放标记，之后删除自身的标记。这便是 “ 延迟标记 ” 的思想。

我们给 $a$ 数组多一个词条 $sign$，用来存储所含的标记。并写一个标记下放的函数：

```cpp
//这里以最小值的标记下放为例，下同
void down(int p)
{
    if (!a[p].sign) return;//结点不存在标记，结束
    a[p<<1].mi+=a[p].sign;//修改左子结点值
    a[p<<1].sign+=a[p].sign;//标记下放至左子结点
    a[p<<1|1].mi+=a[p].sign;//修改右子结点值
    a[p<<1|1].sign+=a[p].sign;//标记下放至右子结点
    a[p].sign=0;//清除自身标记
}
```

同时修改 `change` 函数，我们直接将之前的单点修改改为区间修改，这样我们只需要对一个长度为 $1$ 的区间进行修改就等同于单点修改。

```cpp
void change(int p, int l, int r, int s, int t, ll k)
{
    if(l >= s && r <= t)
    {
        a[p].mi += k;
        a[p].sign += k;
        return ;
    }
    down(p);
    if (s <= mid) change(p << 1,l,mid,s,t,k);
    if (t > mid) change(p<<1|1,mid+1,r,s,t,k);
    update(p);
}
```
